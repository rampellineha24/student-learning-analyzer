<script>

document.addEventListener("DOMContentLoaded", function () {

    // SIDEBAR NAVIGATION
    const navLinks = document.querySelectorAll(".sidebar .nav-link");

    navLinks.forEach(function (link) {
        link.addEventListener("click", function (event) {

            const href = link.getAttribute("href");

            if (href === "{{ url_for('dashboard') }}") {
                return;
            }

            if (href && href.startsWith("#")) {
                event.preventDefault();

                const target = document.querySelector(href);

                if (target) {
                    target.scrollIntoView({
                        behavior: "smooth",
                        block: "start"
                    });

                    navLinks.forEach(function (item) {
                        item.classList.remove("active");
                    });

                    link.classList.add("active");

                    history.replaceState(null, "", href);
                }
            }
        });
    });

    // RESTORE SECTION FROM URL
    function restoreNavigation() {
        const hash = window.location.hash;

        if (hash) {
            const target = document.querySelector(hash);
            const activeLink = document.querySelector(
                '.sidebar .nav-link[href="' + hash + '"]'
            );

            if (target) {
                target.scrollIntoView({ block: "start" });
            }

            if (activeLink) {
                navLinks.forEach(link => link.classList.remove("active"));
                activeLink.classList.add("active");
            }
        }
    }

    if (window.location.hash) {
        setTimeout(restoreNavigation, 100);
    }

    // OUTCOME DISTRIBUTION CHART
    const outcomeCanvas = document.getElementById("outcomeChart");

    if (outcomeCanvas && typeof Chart !== "undefined") {
        const outcomeLabels = JSON.parse(outcomeCanvas.dataset.labels);
        const outcomeValues = JSON.parse(outcomeCanvas.dataset.values);

        new Chart(outcomeCanvas, {
            type: "doughnut",
            data: {
                labels: outcomeLabels,
                datasets: [{
                    data: outcomeValues,
                    backgroundColor: [
                        "#26966d",
                        "#e3a33a",
                        "#6478ce",
                        "#d66a6a"
                    ],
                    borderWidth: 2,
                    borderColor: "#ffffff"
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        position: "bottom"
                    }
                }
            }
        });
    }

    // FEATURE IMPORTANCE CHART
    const importanceCanvas = document.getElementById("importanceChart");

    if (importanceCanvas && typeof Chart !== "undefined") {
        const importanceLabels = JSON.parse(importanceCanvas.dataset.labels);
        const importanceValues = JSON.parse(importanceCanvas.dataset.values);

        new Chart(importanceCanvas, {
            type: "bar",
            data: {
                labels: importanceLabels,
                datasets: [{
                    label: "Importance Mean",
                    data: importanceValues,
                    backgroundColor: "#6478ce",
                    borderRadius: 5
                }]
            },
            options: {
                indexAxis: "y",
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: {
                        display: false
                    }
                },
                scales: {
                    x: {
                        title: {
                            display: true,
                            text: "Decrease in Macro F1"
                        }
                    }
                }
            }
        });
    }

});

</script>

document.addEventListener("DOMContentLoaded", function () {

    // Set the width of each prediction probability bar.
    const progressBars = document.querySelectorAll(".progress-fill");

    progressBars.forEach(function (bar) {
        const probability = Number(bar.dataset.width);

        if (Number.isFinite(probability)) {
            const safeWidth = Math.min(100, Math.max(0, probability));
            bar.style.width = safeWidth + "%";
        }
    });

});
